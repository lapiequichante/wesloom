cmake_minimum_required(VERSION 3.16)
project(wxsl_dawn LANGUAGES C CXX)

option(WXSL_BUILD_RENDERER "Build the offline Dawn renderer" OFF)
option(WXSL_BUILD_ABI_TESTS "Build the dynamic C ABI harness" ON)
if(WXSL_BUILD_RENDERER)
    find_package(Threads REQUIRED)
    find_package(Dawn CONFIG REQUIRED)
    find_package(nlohmann_json CONFIG REQUIRED)
    find_package(stb CONFIG REQUIRED)
    add_library(wxsl_compiler_cpp src/compiler.cpp)
    target_include_directories(wxsl_compiler_cpp PUBLIC include PRIVATE ../crates/wxsl-ffi/include)
    target_compile_features(wxsl_compiler_cpp PUBLIC cxx_std_20)
    target_link_libraries(wxsl_compiler_cpp PUBLIC nlohmann_json::nlohmann_json PRIVATE ${CMAKE_DL_LIBS})
    add_library(wxsl_render_cpp src/renderer.cpp)
    target_include_directories(wxsl_render_cpp PUBLIC include)
    target_compile_features(wxsl_render_cpp PUBLIC cxx_std_20)
    target_link_libraries(wxsl_render_cpp PUBLIC nlohmann_json::nlohmann_json wxsl_compiler_cpp PRIVATE dawn::webgpu_dawn)
    add_executable(pbr_cube_cpp src/pbr_cube.cpp)
    target_link_libraries(pbr_cube_cpp PRIVATE wxsl_render_cpp wxsl_compiler_cpp stb::stb)
    add_executable(renderer_smoke tests/renderer_smoke.cpp)
    target_include_directories(renderer_smoke PRIVATE src)
    target_link_libraries(renderer_smoke PRIVATE wxsl_render_cpp stb::stb)
    add_executable(runtime_smoke tests/runtime_smoke.cpp)
    target_include_directories(runtime_smoke PRIVATE src)
    target_link_libraries(runtime_smoke PRIVATE wxsl_render_cpp)
    if(MSVC)
        target_compile_options(wxsl_render_cpp PRIVATE /W4 /WX)
        target_compile_options(wxsl_compiler_cpp PRIVATE /W4 /WX)
    else()
        target_compile_options(wxsl_render_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_compile_options(wxsl_compiler_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()

if(WXSL_BUILD_ABI_TESTS)
set(WXSL_FFI_LIBRARY "" CACHE FILEPATH "Built wxsl-ffi cdylib")
set(WXSL_ABI_FIXTURES "" CACHE PATH "Native Rust shader parity fixtures")
if(NOT EXISTS "${WXSL_FFI_LIBRARY}" OR NOT IS_DIRECTORY "${WXSL_ABI_FIXTURES}")
    message(FATAL_ERROR "Set WXSL_FFI_LIBRARY and WXSL_ABI_FIXTURES; see dawn/README.md")
endif()

enable_testing()
foreach(language c cpp)
    add_executable(abi_smoke_${language} tests/abi_smoke.${language})
    target_include_directories(abi_smoke_${language} PRIVATE ../crates/wxsl-ffi/include include)
    target_link_libraries(abi_smoke_${language} PRIVATE ${CMAKE_DL_LIBS})
    if(MSVC)
        target_compile_options(abi_smoke_${language} PRIVATE /W4 /WX)
    else()
        target_compile_options(abi_smoke_${language} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME abi_${language} COMMAND abi_smoke_${language}
        "${WXSL_FFI_LIBRARY}" "${CMAKE_CURRENT_SOURCE_DIR}/../crates/wxsl-frame/assets/presets"
        "${WXSL_ABI_FIXTURES}")
endforeach()
set_target_properties(abi_smoke_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
set_target_properties(abi_smoke_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
endif()
